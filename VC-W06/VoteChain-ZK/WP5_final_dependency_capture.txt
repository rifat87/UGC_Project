   159	const updateZkRegistrationStatus = async (req, res) => {
   160	  try {
   161	    const { identityCommitment, registrationStatus, transactionHash } = req.body
   162	    if (!/^\d+$/.test(String(identityCommitment || ''))) {
   163	      return res.status(400).json({ success: false, message: 'Valid identityCommitment is required' })
   164	    }
   165	    if (!['registered', 'failed'].includes(registrationStatus)) {
   166	      return res.status(400).json({ success: false, message: 'registrationStatus must be registered or failed' })
   167	    }
   168	
   169	    const voter = await Voter.findOne({ zkIdentityCommitment: String(identityCommitment) })
   170	    if (!voter) return res.status(404).json({ success: false, message: 'Voter enrollment not found' })
   171	
   172	    const contract = getContract()
   173	    const onChain = await contract.registeredCommitments(BigInt(identityCommitment))
   174	
   175	    if (registrationStatus === 'registered') {
   176	      if (!onChain) {
   177	        return res.status(409).json({ success: false, message: 'Commitment is not registered in the active ZK contract' })
   178	      }
   179	      const groupIndex = Number(await contract.getCommitmentIndex(BigInt(identityCommitment)))
   180	      voter.zkRegistrationStatus = 'registered'
   181	      voter.zkGroupIndex = groupIndex
   182	      if (transactionHash) voter.zkRegistrationTxHash = String(transactionHash)
   183	    } else {
   184	      // Never let an API caller downgrade a commitment that the blockchain already
   185	      // confirms as a member. Blockchain membership is the authority boundary.
   186	      if (onChain) {
   187	        return res.status(409).json({ success: false, message: 'Commitment is already registered on-chain and cannot be marked failed' })
   188	      }
   189	      voter.zkRegistrationStatus = 'failed'
   190	      voter.zkGroupIndex = undefined
   191	      if (transactionHash) voter.zkRegistrationTxHash = String(transactionHash)
   192	    }
   193	
   194	    await voter.save()
   195	    return res.json({
   196	      success: true,
   197	      data: {
   198	        identityCommitment: voter.zkIdentityCommitment,
   199	        registrationStatus: voter.zkRegistrationStatus,
   200	        groupIndex: voter.zkGroupIndex ?? null,
   201	      },
   202	    })
   203	  } catch (error) {
   204	    console.error('Voter ZK registration reconciliation failed:', error.message)
   205	    return res.status(500).json({ success: false, message: 'Failed to reconcile voter commitment registration' })
   206	  }
   207	}

     1	import crypto from 'node:crypto'
     2	
     3	const ALGORITHM = 'aes-256-gcm'
     4	const IV_BYTES = 12
     5	const CREDENTIAL_VERSION = 1
     6	
     7	function getCredentialKey(secret = process.env.ZK_CREDENTIAL_SECRET_KEY) {
     8	  if (!secret) throw new Error('ZK_CREDENTIAL_SECRET_KEY is missing from environment')
     9	  if (Buffer.byteLength(secret, 'utf8') !== 32) {
    10	    throw new Error('ZK_CREDENTIAL_SECRET_KEY must be exactly 32 bytes for AES-256-GCM')
    11	  }
    12	  return Buffer.from(secret, 'utf8')
    13	}
    14	
    15	export function encryptZkCredential(credential, identityCommitment, secret) {
    16	  if (typeof credential !== 'string' || credential.length === 0) throw new Error('Semaphore credential is required')
    17	  if (!identityCommitment) throw new Error('Identity commitment is required for credential binding')
    18	
    19	  const key = getCredentialKey(secret)
    20	  const iv = crypto.randomBytes(IV_BYTES)
    21	  const cipher = crypto.createCipheriv(ALGORITHM, key, iv)
    22	  cipher.setAAD(Buffer.from(String(identityCommitment), 'utf8'))
    23	
    24	  const ciphertext = Buffer.concat([cipher.update(credential, 'utf8'), cipher.final()])
    25	  const authTag = cipher.getAuthTag()
    26	
    27	  return {
    28	    version: CREDENTIAL_VERSION,
    29	    iv: iv.toString('hex'),
    30	    ciphertext: ciphertext.toString('hex'),
    31	    authTag: authTag.toString('hex'),
    32	  }
    33	}
    34	
    35	export function decryptZkCredential(encryptedCredential, identityCommitment, secret) {
    36	  if (!encryptedCredential || encryptedCredential.version !== CREDENTIAL_VERSION) {
    37	    throw new Error('Unsupported or missing ZK credential version')
    38	  }
    39	  if (!identityCommitment) throw new Error('Identity commitment is required for credential binding')
    40	
    41	  const key = getCredentialKey(secret)
    42	  const decipher = crypto.createDecipheriv(ALGORITHM, key, Buffer.from(encryptedCredential.iv, 'hex'))
    43	  decipher.setAAD(Buffer.from(String(identityCommitment), 'utf8'))
    44	  decipher.setAuthTag(Buffer.from(encryptedCredential.authTag, 'hex'))
    45	
    46	  const plaintext = Buffer.concat([
    47	    decipher.update(Buffer.from(encryptedCredential.ciphertext, 'hex')),
    48	    decipher.final(),
    49	  ])
    50	  return plaintext.toString('utf8')
    51	}

   383	            var rb = 0;
   384	            rb < 253;
   385	            rb++
   386	        ) {
   387	            rRange[j].in[rb] <==
   388	                rBits[j].out[rb];
   389	        }
   390	
   391	        rRange[j].in[253] <==
   392	            0;
   393	
   394	        rRange[j].out ===
   395	            0;
   396	
   397	        /*
   398	         * R_j = r_j G
   399	         */
   400	        rG[j] =
   401	            EscalarMulFix(
   402	                253,
   403	                G
   404	            );
   405	
   406	        for (
   407	            var rg = 0;
   408	            rg < 253;
   409	            rg++
   410	        ) {
   411	            rG[j].e[rg] <==
   412	                rBits[j].out[rg];
   413	        }
   414	
   415	        Rx[j] <==
   416	            rG[j].out[0];
   417	
   418	        Ry[j] <==
   419	            rG[j].out[1];
   420	
   421	        /*
   422	         * r_j Y
   423	         */
   424	        rY[j] =
   425	            EscalarMulAny(253);
   426	
   427	        for (
   428	            var ry = 0;
   429	            ry < 253;
   430	            ry++
   431	        ) {
   432	            rY[j].e[ry] <==
   433	                rBits[j].out[ry];
   434	        }
   435	
   436	        rY[j].p[0] <==
   437	            Yx;
   438	
   439	        rY[j].p[1] <==
   440	            Yy;
   441	
   442	        /*
   443	         * Encode bit as:
   444	         *   0 -> identity (0,1)
   445	         *   1 -> G
   446	         */
   447	        messageX[j] <==
   448	            bit[j] *
   449	            G[0];
   450	
   451	        messageY[j] <==
   452	            1 +
   453	            bit[j] *
   454	            (
   455	                G[1] - 1
   456	            );
   457	
   458	        /*
   459	         * S_j = r_j Y + bit_j G
   460	         */
   461	        addPoint[j] =
   462	            BabyAdd();
   463	
   464	        addPoint[j].x1 <==
   465	            rY[j].out[0];
   466	
   467	        addPoint[j].y1 <==
   468	            rY[j].out[1];
   469	
   470	        addPoint[j].x2 <==
   471	            messageX[j];
   472	
   473	        addPoint[j].y2 <==
   474	            messageY[j];
   475	
   476	        Sx[j] <==
   477	            addPoint[j].xout;
   478	
   479	        Sy[j] <==
   480	            addPoint[j].yout;
   481	    }
   482	
   483	    /*
   484	     * Exactly one active candidate.
   485	     */
   486	    bitSum[MAX_CANDIDATES] ===
   487	        1;
   488	}

    33	 * a*x^2 + y^2 = 1 + d*x^2*y^2

    41	    uint256 internal constant P =
    42	        21888242871839275222246405745257275088548364400416034343698204186575808495617;

    56	        uint256 x,
    57	        uint256 y
    58	    )
    59	        private
    60	        pure
    61	        returns (uint256)
    62	    {
    63	        return
    64	            x >= y
    65	                ? x - y
    66	                : P - (y - x);

    71	     *
    72	     * inverse(a) = a^(P-2) mod P
    73	     */
    74	    function _inverse(
    75	        uint256 x
    76	    )
    77	        private
    78	        view
    79	        returns (uint256 out)
    80	    {
    81	        require(
    82	            x != 0,
    83	            "BABYJUB_ZERO_DENOMINATOR"
    84	        );

    86	        bytes memory input =
    87	            abi.encode(
    88	                uint256(32),
    89	                uint256(32),
    90	                uint256(32),
    91	                x,
    92	                P - 2,
    93	                P
    94	            );

    96	        bytes memory output =
    97	            new bytes(32);
    98	
    99	        bool ok;
   100	
   101	        assembly ("memory-safe") {
   102	            ok := staticcall(
   103	                gas(),
   104	                5,
   105	                add(input, 32),
   106	                mload(input),
   107	                add(output, 32),
   108	                32
   109	            )
   110	        }
   111	
   112	        require(
   113	            ok,
   114	            "MODEXP_FAILED"
   115	        );
   116

   118	            abi.decode(
   119	                output,

   129	    function add(
   130	        Point memory p1,
   131	        Point memory p2
   132	    )
   133	        internal
   134	        view
   135	        returns (Point memory r)
   136	    {
   137	        require(
   138	            p1.x < P &&
   139	            p1.y < P &&
   140	            p2.x < P &&
   141	            p2.y < P,
   142	            "NON_CANONICAL_POINT"
   143	        );

   145	        uint256 x1x2 =
   146	            mulmod(
   147	                p1.x,
   148	                p2.x,
   149	                P
   150	            );
   151	
   152	        uint256 y1y2 =
   153	            mulmod(
   154	                p1.y,
   155	                p2.y,
   156	                P
   157	            );
   158	
   159	        uint256 x1y2 =
   160	            mulmod(
   161	                p1.x,
   162	                p2.y,
   163	                P
   164	            );
   165	
   166	        uint256 y1x2 =
   167	            mulmod(
   168	                p1.y,
   169	                p2.x,
   170	                P
   171	            );
   172	
   173	        uint256 t =
   174	            mulmod(
   175	                x1x2,
   176	                y1y2,
   177	                P
   178	            );
   179	
   180	        uint256 dt =
   181	            mulmod(
   182	                D,
   183	                t,
   184	                P
   185	            );
   186	
   187	        uint256 numeratorX =
   188	            addmod(
   189	                x1y2,
   190	                y1x2,
   191	                P
   192	            );
   193	
   194	        uint256 numeratorY =
   195	            _submod(
   196	                y1y2,
   197	                mulmod(
   198	                    A,
   199	                    x1x2,
   200	                    P
   201	                )
   202	            );
   203	
   204	        uint256 denominatorX =
   205	            addmod(
   206	                1,
   207	                dt,
   208	                P
   209	            );
   210	
   211	        uint256 denominatorY =
   212	            _submod(
   213	                1,
   214	                dt
   215	            );
   216	
   217	        r.x =
   218	            mulmod(
   219	                numeratorX,
   220	                _inverse(
   221	                    denominatorX
   222	                ),
   223	                P
   224	            );
   225	
   226	        r.y =
   227	            mulmod(
   228	                numeratorY,
   229	                _inverse(
   230	                    denominatorY
   231	                ),
   232	                P
   233	            );
   234	    }
