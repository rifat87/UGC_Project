import { useCallback, useEffect, useState } from "react"
import { useContract } from "@/hooks/useContract"
import { useWallet } from "@/components/ui/wallet-provider"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Button } from "@/components/ui/button"
import { Alert, AlertDescription } from "@/components/ui/alert"
import { Badge } from "@/components/ui/badge"
import { Progress } from "@/components/ui/progress"
import { useToast } from "@/hooks/use-toast"
import { Vote, CheckCircle2, User, MapPin, Loader2, AlertCircle, RefreshCw } from "lucide-react"
import { VerificationModal } from "@/components/Biometric/VerificationModal"
import { getPublicElectionData, type PublicCandidate, type PublicElectionData, type ZkVotePreparation } from "@/services/api"

export function VoteCasting() {
  const { isWalletChecked, isConnected, address } = useWallet()
  const { isInitialized, isAdmin, castVote, isLoading: contractLoading, error: contractError } = useContract()
  const { toast } = useToast()
  const [data, setData] = useState<PublicElectionData | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  const [selectedCandidate, setSelectedCandidate] = useState<number | null>(null)
  const [isVoting, setIsVoting] = useState(false)
  const [votedCandidate, setVotedCandidate] = useState<PublicCandidate | null>(null)
  const [transactionHash, setTransactionHash] = useState<string | null>(null)
  const [isModalOpen, setIsModalOpen] = useState(false)
  const [pendingCandidate, setPendingCandidate] = useState<number | null>(null)

  const refresh = useCallback(async () => {
    setLoading(true)
    setError(null)
    try {
      setData(await getPublicElectionData())
    } catch (requestError) {
      setError(requestError instanceof Error ? requestError.message : "Failed to fetch election data")
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    void refresh()
  }, [refresh])

  const handleVote = (candidateId: number) => {
    if (!isConnected) {
      toast({ title: "Wallet Not Connected", description: "Connect the vote-submission wallet before voting.", variant: "destructive" })
      return
    }
    if (isAdmin) {
      toast({ title: "Admin wallet cannot use voter UI", description: "Use a non-admin vote-submission wallet.", variant: "destructive" })
      return
    }
    if (data?.electionState !== "Active") {
      toast({ title: "Voting unavailable", description: `Voting is only available while the election is Active. Current state: ${data?.electionState || "unknown"}.`, variant: "destructive" })
      return
    }
    setPendingCandidate(candidateId)
    setIsModalOpen(true)
  }

  const handleVerified = async (preparation: ZkVotePreparation) => {
    if (pendingCandidate === null || !data) return
    const candidateId = pendingCandidate
    const candidate = data.candidates.find((item) => item.candidateId === candidateId) || null

    try {
      setIsVoting(true)
      setSelectedCandidate(candidateId)
      if (preparation.candidateId !== candidateId) throw new Error("Prepared proof candidate does not match selected candidate")
      const receipt = await castVote(candidateId, preparation.proof)
      setVotedCandidate(candidate)
      setTransactionHash(receipt?.hash || null)
      toast({ title: "Vote Cast Successfully", description: `Vote for ${candidate?.name || `candidate #${candidateId}`} was recorded on-chain.` })
      await refresh()
    } catch (voteError) {
      console.error("Vote transaction failed", voteError)
      toast({ title: "Vote Failed", description: voteError instanceof Error ? voteError.message : "There was an error casting the vote.", variant: "destructive" })
    } finally {
      setIsVoting(false)
      setSelectedCandidate(null)
      setPendingCandidate(null)
      setIsModalOpen(false)
    }
  }

  const modal = (
    <VerificationModal open={isModalOpen} candidateId={pendingCandidate} onClose={() => { setIsModalOpen(false); setPendingCandidate(null) }} onVerified={handleVerified} />
  )

  if (!isWalletChecked || !isConnected || !isInitialized || loading || contractLoading) {
    return <>{modal}<div className="container mx-auto p-4 max-w-4xl"><Card><CardHeader><CardTitle className="flex items-center gap-2"><Vote className="h-6 w-6" />Cast Your Vote</CardTitle></CardHeader><CardContent className="flex items-center justify-center py-12"><Loader2 className="h-8 w-8 animate-spin" /></CardContent></Card></div></>
  }

  if (error || contractError || !data) {
    return <>{modal}<div className="container mx-auto p-4 max-w-4xl"><Card><CardHeader><CardTitle>Cast Your Vote</CardTitle></CardHeader><CardContent className="text-center py-12"><AlertCircle className="h-8 w-8 mx-auto text-destructive" /><p className="text-destructive mt-3">{error || contractError?.message || "Election data unavailable"}</p><Button onClick={refresh} variant="outline" className="mt-4"><RefreshCw className="mr-2 h-4 w-4" />Try Again</Button></CardContent></Card></div></>
  }

  if (votedCandidate) {
    return <>{modal}<div className="container mx-auto p-4 max-w-2xl"><Card><CardHeader className="text-center"><div className="flex items-center justify-center gap-2 text-green-600 mb-2"><CheckCircle2 className="h-8 w-8" /><CardTitle className="text-2xl">Vote Recorded</CardTitle></div></CardHeader><CardContent className="text-center space-y-4"><p>Your vote for <strong>{votedCandidate.name}</strong> was recorded on the blockchain.</p>{transactionHash && <Badge variant="secondary" className="break-all">Transaction: {transactionHash}</Badge>}</CardContent></Card></div></>
  }

  const totalVotes = data.candidates.reduce((sum, candidate) => sum + candidate.voteCount, 0)
  const votingEnabled = data.electionState === "Active" && !isAdmin

  return (
    <>
      {modal}
      <div className="container mx-auto p-4 max-w-4xl">
        <Card className="mb-6">
          <CardHeader><CardTitle className="flex items-center gap-2"><Vote className="h-6 w-6" />Cast Your Vote <Badge variant={data.electionState === "Active" ? "default" : "secondary"}>{data.electionState}</Badge></CardTitle></CardHeader>
          <CardContent><Alert><AlertDescription>{votingEnabled ? "Select a candidate. Biometric verification gates backend proof generation; the blockchain receives a Semaphore membership proof without exposing a persistent voter identifier." : "Voting is disabled unless the election is Active and a non-admin vote-submission wallet is connected."}</AlertDescription></Alert></CardContent>
        </Card>

        {data.candidates.length === 0 ? (
          <Card><CardContent className="flex items-center justify-center py-12"><div className="text-center space-y-4"><User className="h-8 w-8 mx-auto text-muted-foreground" /><p className="text-muted-foreground">No on-chain registered candidates with linked metadata were found.</p></div></CardContent></Card>
        ) : (
          <div className="grid gap-6">
            {data.candidates.map((candidate) => {
              const percentage = totalVotes > 0 ? (candidate.voteCount / totalVotes) * 100 : 0
              const isCurrentlyVoting = isVoting && selectedCandidate === candidate.candidateId
              return (
                <Card key={candidate.candidateId} className="transition-all hover:shadow-md">
                  <CardHeader>
                    <div className="flex items-start justify-between">
                      <div className="space-y-2">
                        <CardTitle className="flex items-center gap-2"><User className="h-5 w-5" />{candidate.name}</CardTitle>
                        <div className="text-sm text-muted-foreground">Candidate #{candidate.candidateId} · {candidate.party}</div>
                        <div className="flex items-center gap-2 text-muted-foreground"><MapPin className="h-4 w-4" /><span>{candidate.location}</span></div>
                      </div>
                      <div className="text-right"><div className="text-2xl font-bold">{candidate.voteCount}</div><div className="text-sm text-muted-foreground">votes</div></div>
                    </div>
                  </CardHeader>
                  <CardContent className="space-y-4">
                    <div className="space-y-2"><div className="flex justify-between text-sm"><span>Current Support</span><span>{percentage.toFixed(1)}%</span></div><Progress value={percentage} className="h-2" /></div>
                    <Button onClick={() => handleVote(candidate.candidateId)} disabled={!votingEnabled || isVoting} className="w-full" size="lg">
                      {isCurrentlyVoting ? <><Loader2 className="mr-2 h-4 w-4 animate-spin" />Processing Vote...</> : <><Vote className="mr-2 h-4 w-4" />Vote for {candidate.name}</>}
                    </Button>
                  </CardContent>
                </Card>
              )
            })}
          </div>
        )}

        <Card className="mt-6"><CardContent className="pt-6"><div className="flex items-center justify-between text-sm text-muted-foreground"><span>Connected Wallet: {address?.slice(0, 6)}...{address?.slice(-4)}</span><span>Total Votes: {totalVotes}</span></div></CardContent></Card>
      </div>
    </>
  )
}

import { useState } from "react"
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog"
import { Button } from "@/components/ui/button"
import { Loader2, ScanFace, Fingerprint } from "lucide-react"
import { useToast } from "@/hooks/use-toast"
import { prepareZkVote, type ZkVotePreparation } from "@/services/api"

interface VerificationModalProps {
  open: boolean
  candidateId: number | null
  onClose: () => void
  onVerified: (preparation: ZkVotePreparation) => void
}

export function VerificationModal({ open, candidateId, onClose, onVerified }: VerificationModalProps) {
  const { toast } = useToast()
  const [isVerifying, setIsVerifying] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const runVerification = async (path: string, label: string) => {
    if (!candidateId) return
    setError(null)
    setIsVerifying(true)
    try {
      const preparation = await prepareZkVote(path, candidateId)
      toast({ title: `${label} Verified`, description: "Preparing anonymous eligibility proof completed." })
      onVerified(preparation)
      onClose()
    } catch (verificationError) {
      setError(verificationError instanceof Error ? verificationError.message : "Unknown verification error")
    } finally {
      setIsVerifying(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={(value) => !value && onClose()}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Biometric Verification</DialogTitle>
          <DialogDescription>
            Verify the voter. After a successful match, the backend uses the encrypted Semaphore credential internally to prepare an anonymous eligibility proof.
          </DialogDescription>
        </DialogHeader>
        {error && <p className="text-sm text-destructive mb-2">{error}</p>}
        {isVerifying && <p className="text-sm text-muted-foreground">Preparing anonymous eligibility proof...</p>}
        <div className="flex flex-col gap-4">
          <Button disabled={isVerifying || !candidateId} onClick={() => runVerification("/api/votes/verify-face", "Face")} className="w-full flex items-center gap-2">
            {isVerifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ScanFace className="h-4 w-4" />}Verify with Face
          </Button>
          <Button variant="outline" disabled={isVerifying || !candidateId} onClick={() => runVerification("/api/biometric/fingerprint/detect", "Fingerprint")} className="w-full flex items-center gap-2">
            {isVerifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <Fingerprint className="h-4 w-4" />}Verify with Fingerprint
          </Button>
        </div>
        <DialogFooter className="mt-4"><Button variant="ghost" onClick={onClose} disabled={isVerifying}>Cancel</Button></DialogFooter>
      </DialogContent>
    </Dialog>
  )
}

import { useCallback, useEffect, useRef, useState } from "react"
import { ethers } from "ethers"
import { createContract } from "@/config/contract"
import { useWallet } from "@/components/ui/wallet-provider"
import type { ElectionState, SemaphoreProofPayload } from "@/services/api"

const ELECTION_STATES: Record<number, ElectionState> = {
  0: "Registration",
  1: "Active",
  2: "Ended",
}

function toElectionState(value: bigint | number): ElectionState {
  const state = ELECTION_STATES[Number(value)]
  if (!state) throw new Error(`Unknown election state: ${value.toString()}`)
  return state
}

export function useContract() {
  const [contract, setContract] = useState<ethers.Contract | null>(null)
  const [isLoading, setIsLoading] = useState(false)
  const [error, setError] = useState<Error | null>(null)
  const [isInitialized, setIsInitialized] = useState(false)
  const [isAdmin, setIsAdmin] = useState(false)
  const { isConnected, signer, address } = useWallet()
  const lastAddress = useRef<string | null>(null)

  useEffect(() => {
    if (!isConnected || !address || !signer) {
      setContract(null)
      setIsInitialized(false)
      setIsAdmin(false)
      setIsLoading(false)
      lastAddress.current = null
      return
    }

    if (lastAddress.current === address) return
    lastAddress.current = address
    setIsLoading(true)
    setError(null)

    const initializeContract = async () => {
      try {
        const contractInstance = createContract(signer)
        const commission = await contractInstance.electionCommission()
        setContract(contractInstance)
        setIsInitialized(commission !== ethers.ZeroAddress)
        setIsAdmin(address.toLowerCase() === commission.toLowerCase())
      } catch (initializationError) {
        setError(initializationError as Error)
        setContract(null)
        setIsInitialized(false)
        setIsAdmin(false)
      } finally {
        setIsLoading(false)
      }
    }

    void initializeContract()
  }, [isConnected, signer, address])

  const requireContract = useCallback(() => {
    if (!contract) throw new Error("Contract not initialized")
    return contract
  }, [contract])

  const registerCandidate = useCallback(async (candidateId: number) => {
    const instance = requireContract()
    const tx = await instance.registerCandidate(candidateId)
    const receipt = await tx.wait()
    if (receipt?.status === 0) throw new Error("Candidate registration transaction failed")
    return receipt
  }, [requireContract])

  const registerVoterCommitment = useCallback(async (identityCommitment: string) => {
    if (!/^\d+$/.test(identityCommitment)) throw new Error("Invalid Semaphore identity commitment")
    const instance = requireContract()
    const tx = await instance.registerVoterCommitment(BigInt(identityCommitment))
    const receipt = await tx.wait()
    if (receipt?.status === 0) throw new Error("Voter commitment registration transaction failed")
    return receipt
  }, [requireContract])

  const castVote = useCallback(async (candidateId: number, proof: SemaphoreProofPayload) => {
    if (!Number.isSafeInteger(candidateId) || candidateId <= 0) throw new Error("Invalid candidate ID")
    if (BigInt(proof.message) !== BigInt(candidateId)) throw new Error("Proof is not bound to selected candidate")
    if (!Array.isArray(proof.points) || proof.points.length !== 8) throw new Error("Invalid Semaphore proof")

    const instance = requireContract()
    const tx = await instance.castVote(candidateId, proof)
    const receipt = await tx.wait()
    if (receipt?.status === 0) throw new Error("Vote transaction failed")
    return receipt
  }, [requireContract])

  const startElection = useCallback(async () => {
    const instance = requireContract()
    const tx = await instance.startElection()
    return tx.wait()
  }, [requireContract])

  const endElection = useCallback(async () => {
    const instance = requireContract()
    const tx = await instance.endElection()
    return tx.wait()
  }, [requireContract])

  const getElectionState = useCallback(async () => {
    const instance = requireContract()
    return toElectionState(await instance.getElectionState())
  }, [requireContract])

  const getVoteCount = useCallback(async (candidateId: number) => {
    const instance = requireContract()
    return Number(await instance.getVoteCount(candidateId))
  }, [requireContract])

  return {
    contract,
    isLoading,
    error,
    isInitialized,
    isAdmin,
    registerCandidate,
    registerVoterCommitment,
    castVote,
    startElection,
    endElection,
    getElectionState,
    getVoteCount,
  }
}

import { apiUrl } from "@/config/api"

export type ElectionState = "Registration" | "Active" | "Ended"

export interface PublicCandidate {
  candidateId: number
  name: string
  party: string
  location: string
  voteCount: number
}

export interface PublicElectionData {
  electionState: ElectionState
  candidateCount: number
  candidates: PublicCandidate[]
}

interface ApiEnvelope<T> {
  success: boolean
  data?: T
  message?: string
  errors?: string[]
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(apiUrl(path), init)
  const payload = (await response.json()) as ApiEnvelope<T>

  if (!response.ok || payload.success === false) {
    const detail = payload.errors?.join(", ") || payload.message || `Request failed with ${response.status}`
    throw new Error(detail)
  }

  if (payload.data === undefined) {
    throw new Error("API response did not contain data")
  }

  return payload.data
}

export async function getPublicElectionData() {
  return request<PublicElectionData>("/api/elections/public/election-data")
}

export interface CandidateMetadataInput {
  nationalId: string
  name: string
  party: string
  location: string
  age: number
}

export interface CandidateAdminRecord {
  id: string
  candidateId: number
  name: string
  party: string
  location: string
  age: number
  registrationStatus: "pending" | "registered" | "failed"
}

export async function createCandidateMetadata(metadata: CandidateMetadataInput) {
  return request<CandidateAdminRecord>("/api/candidates/register", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(metadata),
  })
}

export async function updateCandidateRegistrationStatus(
  candidateId: number,
  registrationStatus: "pending" | "registered" | "failed",
) {
  return request<{ candidateId: number; registrationStatus: string }>(
    `/api/candidates/${candidateId}/registration-status`,
    {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ registrationStatus }),
    },
  )
}

export interface CreateVoterInput {
  nationalId: string
  name: string
  dateOfBirth: string
  location: string
}

export interface VoterCreationResult {
  identityCommitment: string
}

export interface SemaphoreProofPayload {
  merkleTreeDepth: number
  merkleTreeRoot: string
  nullifier: string
  message: string
  scope: string
  points: [string, string, string, string, string, string, string, string]
}

export interface ZkVotePreparation {
  candidateId: number
  electionId: string
  proof: SemaphoreProofPayload
}

export async function createVoter(input: CreateVoterInput) {
  const response = await fetch(apiUrl("/api/voters"), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(input),
  })
  const payload = await response.json()

  if (!response.ok || !payload.success) throw new Error(payload.message || "Failed to create voter ZK enrollment")
  const identityCommitment = payload.data?.identityCommitment || payload.identityCommitment
  if (!identityCommitment) throw new Error("Identity commitment was not returned from backend")
  return { identityCommitment: String(identityCommitment) } as VoterCreationResult
}

export async function updateVoterZkRegistrationStatus(
  identityCommitment: string,
  registrationStatus: "registered" | "failed",
  transactionHash?: string,
) {
  const response = await fetch(apiUrl("/api/voters/zk-registration-status"), {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ identityCommitment, registrationStatus, transactionHash }),
  })
  const payload = await response.json()
  if (!response.ok || !payload.success) throw new Error(payload.message || "Failed to reconcile voter commitment registration")
  return payload.data as { identityCommitment: string; registrationStatus: string; groupIndex: number | null }
}

export async function prepareZkVote(path: string, candidateId: number) {
  const response = await fetch(apiUrl(path), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ candidateId }),
  })
  const payload = await response.json()
  if (!response.ok || !payload.success) throw new Error(payload.message || "Biometric verification or proof generation failed")
  const data = payload.data || payload
  if (!data?.proof) throw new Error("Anonymous eligibility proof was not returned")
  return data as ZkVotePreparation
}

import { createContext, useContext, useEffect, useState, useRef } from "react"
import { BrowserProvider, JsonRpcSigner } from "ethers"
import { useNavigate } from "react-router-dom"

interface WalletState {
  isConnected: boolean
  address: string | null
  provider: BrowserProvider | null
  signer: JsonRpcSigner | null
  isConnecting: boolean
  isWalletChecked: boolean
}

interface WalletContextType extends WalletState {
  connect: () => Promise<void>
  disconnect: () => Promise<void>
}

const initialState: WalletState = {
  isConnected: false,
  address: null,
  provider: null,
  signer: null,
  isConnecting: false,
  isWalletChecked: false
}

const WalletContext = createContext<WalletContextType | undefined>(undefined)

export function WalletProvider({ children }: { children: React.ReactNode }) {
  const [state, setState] = useState<WalletState>(initialState)
  const mountedRef = useRef(true)
  const connectionRequested = useRef(false)
  const navigate = useNavigate()

  useEffect(() => {
    mountedRef.current = true
    async function checkWallet() {
      if (typeof window.ethereum === 'undefined') {
        setState(prev => ({ ...prev, isWalletChecked: true }))
        return
      }
      try {
        const accounts = await window.ethereum.request({ method: 'eth_accounts' })
        if (accounts.length > 0) {
          const provider = new BrowserProvider(window.ethereum)
          const signer = await provider.getSigner()
          setState({
            isConnected: true,
            address: accounts[0],
            provider,
            signer,
            isConnecting: false,
            isWalletChecked: true
          })
        } else {
          setState(prev => ({ ...prev, isWalletChecked: true }))
        }
      } catch {
        setState(prev => ({ ...prev, isWalletChecked: true }))
      }
    }
    checkWallet()
    return () => {
      mountedRef.current = false
    }
  }, [])

  useEffect(() => {
    if (typeof window.ethereum === 'undefined') {
      return
    }

    const handleAccountsChanged = async (accounts: string[]) => {
      if (!mountedRef.current) return

      if (accounts.length === 0) {
        // User disconnected their wallet
        setState({
          ...initialState,
          isWalletChecked: true
        })
        navigate('/', { replace: true })
      } else {
        // Only update state if we're already connected
        if (state.isConnected) {
          const provider = new BrowserProvider(window.ethereum)
          const signer = await provider.getSigner()
          setState(prev => ({
            ...prev,
            address: accounts[0],
            provider,
            signer
          }))
        }
      }
    }

    const handleChainChanged = () => {
      if (!mountedRef.current) return
      window.location.reload()
    }

    // Set up event listeners
    window.ethereum.on('accountsChanged', handleAccountsChanged)
    window.ethereum.on('chainChanged', handleChainChanged)

    return () => {
      if (window.ethereum) {
        window.ethereum.removeListener('accountsChanged', handleAccountsChanged)
        window.ethereum.removeListener('chainChanged', handleChainChanged)
      }
    }
  }, [state.isConnected])

  const connect = async () => {
    if (typeof window.ethereum === 'undefined') {
      throw new Error('MetaMask is not installed')
    }

    // Prevent multiple simultaneous connection attempts
    if (connectionRequested.current || state.isConnecting) {
      return
    }

    try {
      connectionRequested.current = true
      setState(prev => ({ ...prev, isConnecting: true }))
      
      const accounts = await window.ethereum.request({ method: 'eth_requestAccounts' })
      if (accounts.length > 0) {
        const provider = new BrowserProvider(window.ethereum)
        const signer = await provider.getSigner()
        
        setState({
          isConnected: true,
          address: accounts[0],
          provider,
          signer,
          isConnecting: false,
          isWalletChecked: true
        })
        
        // Role-based routing: Check if user is admin
        try {
          const { createContract } = await import('@/config/contract')
          const contractInstance = createContract(signer)
          const commission = await contractInstance.electionCommission()
          
          if (accounts[0].toLowerCase() === commission.toLowerCase()) {
            // User is admin - redirect to admin dashboard
        navigate('/admin', { replace: true })
          } else {
            // User is voter - redirect to voter dashboard
            navigate('/voter', { replace: true })
          }
        } catch (contractError) {
          console.error('Error determining role:', contractError)
          // If role determination fails, stay on current page
          // User can manually navigate where they need to go
        }
      }
    } catch (error) {
      console.error('Error connecting to MetaMask:', error)
      setState(prev => ({ ...prev, isConnecting: false, isWalletChecked: true }))
      throw error
    } finally {
      connectionRequested.current = false
    }
  }

  const disconnect = async () => {
    try {
      if (window.ethereum) {
        // Revoke all permissions from MetaMask
        await window.ethereum.request({
          method: "wallet_revokePermissions",
          params: [{ eth_accounts: {} }]
        });
      }

      // Reset all state
      setState({
        ...initialState,
        isWalletChecked: true
      })
      
      // Clear any cached data
      localStorage.removeItem('walletConnected')
      
      // Force a page reload to ensure clean state
      window.location.reload()
    } catch (error) {
      console.error('Error disconnecting wallet:', error)
      // Even if there's an error, we still want to reset everything
      setState({
        ...initialState,
        isWalletChecked: true
      })
      localStorage.removeItem('walletConnected')
      window.location.reload()
    }
  }

  return (
    <WalletContext.Provider value={{ ...state, connect, disconnect }}>
      {children}
    </WalletContext.Provider>
  )
}

export function useWallet() {
  const context = useContext(WalletContext)
  if (context === undefined) {
    throw new Error('useWallet must be used within a WalletProvider')
  }
  return context
}

  if (!candidateId) {
    return res.status(400).json({ success: false, message: 'A positive candidateId is required before proof generation' })
  }

  try {
    console.log('[VoteController] Starting biometric-gated ZK proof preparation')
    const faceRecPath = path.join(__dirname, '../../votechain-face-recognition', 'face_rec_demo.py')
    const faceRecDir = path.dirname(faceRecPath)

    if (!fs.existsSync(faceRecPath)) {
      return res.status(500).json({ success: false, message: 'Face recognition script not found' })
    }

    const pythonProcess = spawn('python', [faceRecPath], { cwd: faceRecDir })
    let stdout = ''

    pythonProcess.stdout.on('data', (data) => {
      stdout += data.toString()
    })
    pythonProcess.stderr.on('data', () => {
      console.error('[VoteController] face-recognition process reported an error')
    })

    pythonProcess.on('close', async (code) => {
      try {
        if (code !== 0) {
          return res.status(500).json({ success: false, message: 'Face recognition process failed' })
        }

        const outputLower = stdout.toLowerCase()
        const isSuccess = outputLower.includes('face recognition successful') || outputLower.includes('result = success')
        if (!isSuccess) return res.json({ success: false, message: 'Face not recognised' })

        const idMatch = stdout.match(/Matched ID:\s*([\w-]+)/i)
        const matchedNid = idMatch ? idMatch[1].trim() : null
        if (!matchedNid) {
          return res.status(422).json({ success: false, message: 'Face verification did not return a usable voter identity' })
        }

        const voter = await Voter.findOne({ nationalId: matchedNid })
        if (!voter) return res.status(404).json({ success: false, message: 'Verified voter is not enrolled' })

        const proofPayload = await prepareProofAfterBiometricAuthentication({ biometricAccepted: true, voter, candidateId, proofGenerator: generateVoteProofForVoter })

          success: true,
          message: 'Face verification successful; anonymous eligibility proof prepared',
          data: proofPayload,
        })

        console.error('[VoteController] ZK proof preparation failed:', error.message)
        return res.status(500).json({ success: false, message: 'Unable to prepare anonymous eligibility proof' })
      }
    })

      console.error('[VoteController] Failed to start face recognition:', error.message)

function sendSerialJSON(obj) {
  return new Promise((resolve, reject) => {
    initSerial()

    const jsonStr = JSON.stringify(obj)
    console.log(`[Serial] Sending ${obj.cmd || 'UNKNOWN'} command`)

    const onData = (data) => {
      const clean = data.trim()
      console.log('[Serial] Received device response')
      try {
        const parsed = JSON.parse(clean)
        // Assume ESP32 always responds once per command
        parser.removeListener('data', onData)
        resolve(parsed)
      } catch {
        console.warn("[Serial] Ignored non-JSON data")
      }
    }

    parser.on('data', onData)

    serial.write(jsonStr + "\n", (err) => {
      if (err) {
        parser.removeListener('data', onData)
        console.error("[Serial] Failed to write to serial port:", err.message)
        reject(err)
      } else {
        console.log("[Serial] Command sent successfully.")
      }
    })
  })
}

/**
 * Pure authentication gate placed in front of Semaphore proof generation.
 * The caller injects the backend proof generator only after a successful biometric match,
 * which keeps this boundary testable without loading blockchain/network dependencies.
 */
export async function prepareProofAfterBiometricAuthentication({
  biometricAccepted,
  voter,
  candidateId,
  proofGenerator,
}) {
  if (biometricAccepted !== true) {
    throw new Error('Biometric authentication is required before ZK proof generation')
  }
  if (!voter) {
    throw new Error('Biometrically verified voter record is required before ZK proof generation')
  }
  if (typeof proofGenerator !== 'function') {
    throw new Error('Backend ZK proof generator is required')
  }

  return proofGenerator(voter, candidateId)
}

import { normalizeSemaphoreProof } from './proofSerialization.js'

function validateCandidateId(candidateId) {
  const value = Number(candidateId)
  if (!Number.isSafeInteger(value) || value <= 0) throw new Error('A positive candidateId is required')
  return value
}

export async function generateBoundSemaphoreProof({ runtime, voter, candidateId, electionId, group, chainRegistered, secret, snarkArtifacts }) {
  if (!voter) throw new Error('Verified voter record is required')
  if (chainRegistered !== true) throw new Error('Voter ZK commitment is not registered on-chain')
  if (!voter.zkIdentityCommitment || !voter.zkCredential) throw new Error('Voter does not have an enrolled ZK credential')

  const selectedCandidateId = validateCandidateId(candidateId)
  const normalizedElectionId = BigInt(electionId)
  const exportedCredential = decryptZkCredential(voter.zkCredential, voter.zkIdentityCommitment, secret)
  const identity = runtime.Identity.import(exportedCredential)

  if (identity.commitment.toString() !== voter.zkIdentityCommitment.toString()) {
    throw new Error('Decrypted Semaphore identity does not match stored commitment')
  }

  const proof = await runtime.generateProof(
    identity,
    group,
    BigInt(selectedCandidateId),
    normalizedElectionId,
    undefined,
    snarkArtifacts,
  )
  const normalized = normalizeSemaphoreProof(proof)

  if (BigInt(normalized.message) !== BigInt(selectedCandidateId)) throw new Error('Generated proof message does not match candidateId')
  if (BigInt(normalized.scope) !== normalizedElectionId) throw new Error('Generated proof scope does not match electionId')

  if (typeof runtime.verifyProof === 'function' && !(await runtime.verifyProof(proof))) {
    throw new Error('Generated Semaphore proof failed local verification')
  }

  return {
    candidateId: selectedCandidateId,
    electionId: normalizedElectionId.toString(),
    proof: normalized,
  }
}

import { getContract } from '../../utils/contract.js'
import { loadSemaphoreRuntime } from './semaphoreRuntime.js'
import { reconstructElectionGroup } from './groupService.js'
import { generateBoundSemaphoreProof } from './proofCore.js'
import { getConfiguredSnarkArtifacts } from './snarkArtifacts.js'

export async function generateVoteProofForVoter(voter, candidateId, options = {}) {
  const runtime = options.runtime || await loadSemaphoreRuntime()
  const contract = options.contract || getContract()
  const normalizedCandidateId = Number(candidateId)

  if (!Number.isSafeInteger(normalizedCandidateId) || normalizedCandidateId <= 0) {
    throw new Error('A positive candidateId is required before ZK proof generation')
  }
  if (!voter?.zkIdentityCommitment || !/^\d+$/.test(String(voter.zkIdentityCommitment))) {
    throw new Error('Verified voter does not have a valid ZK identity commitment')
  }

  const [{ group }, electionId, chainRegistered, electionState, candidateState] = await Promise.all([
    reconstructElectionGroup({ runtime, contract, fromBlock: options.fromBlock }),
    contract.electionId(),
    contract.registeredCommitments(BigInt(voter.zkIdentityCommitment)),
    contract.getElectionState(),
    contract.candidates(BigInt(normalizedCandidateId)),
  ])

  if (Number(electionState) !== 1) {
    throw new Error('Anonymous vote proof generation is only available while the election is Active')
  }
  const candidateExists = Boolean(candidateState?.exists ?? candidateState?.[0])
  if (!candidateExists) {
    throw new Error('Selected candidate is not registered on-chain')
  }

  return generateBoundSemaphoreProof({
    runtime,
    voter,
    candidateId: normalizedCandidateId,
    electionId,
    group,
    chainRegistered: Boolean(chainRegistered),
    secret: options.secret,
    snarkArtifacts: options.snarkArtifacts || getConfiguredSnarkArtifacts(),
  })
}

import { getContract } from '../../utils/contract.js'
import { loadSemaphoreRuntime } from './semaphoreRuntime.js'
import { reconstructGroupFromRecords } from './groupCore.js'
import { commitmentRecordsFromRegistrationEvents } from './groupEvents.js'

export async function reconstructElectionGroup({ runtime, contract, fromBlock } = {}) {
  const semaphoreRuntime = runtime || await loadSemaphoreRuntime()
  const zkContract = contract || getContract()
  const startBlock = fromBlock ?? Number(process.env.ZK_CONTRACT_DEPLOYMENT_BLOCK || 0)

  if (!Number.isSafeInteger(Number(startBlock)) || Number(startBlock) < 0) {
    throw new Error('ZK_CONTRACT_DEPLOYMENT_BLOCK must be a non-negative safe integer when configured')
  }

  if (!zkContract.filters?.VoterCommitmentRegistered || typeof zkContract.queryFilter !== 'function') {
    throw new Error('ZK contract event-query interface is required for deterministic group reconstruction')
  }

  // ZKLocalVoting is itself the Semaphore group admin and exposes no update/remove
  // membership operation. Therefore its successful registration events form the
  // authoritative append-only member ordering for this election group.
  const filter = zkContract.filters.VoterCommitmentRegistered()
  const events = await zkContract.queryFilter(filter, Number(startBlock), 'latest')
  const voterRecords = commitmentRecordsFromRegistrationEvents(events)

  const [chainSize, chainRoot] = await Promise.all([
    zkContract.getGroupSize(),
    zkContract.getGroupRoot(),
  ])

  return reconstructGroupFromRecords({
    runtime: semaphoreRuntime,
    voterRecords,
    chainSize,
    chainRoot,
  })
}

router.post('/capture-face', async (req, res) => {
    const { nid } = req.body;

    if (!nid) {
        return res.status(400).json({ success: false, message: 'NID is required' });
    }

    let responseSent = false;

    try {
        const faceRecognitionPath = path.join(__dirname, '../../votechain-face-recognition');
        const pythonProcess = spawn('python', ['dataset.py', nid], {
            cwd: faceRecognitionPath,
            stdio: ['pipe', 'pipe', 'pipe']
        });

        pythonProcess.stdout.on('data', (data) => {
            void data;
            console.log('Face capture process produced output');
        });

        pythonProcess.stderr.on('data', (data) => {
            void data;
            console.error('Face capture process reported an error');
        });

        pythonProcess.on('close', (code) => {
            console.log(`Python process exited with code ${code}`);

            if (!responseSent) {
                if (code === 0) {
                    res.json({
                        success: true,
                        message: 'Face captured successfully'
                    });
                } else {
                    res.status(500).json({
                        success: false,
                        message: 'Face capture failed'
                    });
                }
                responseSent = true;
            }
        });

        pythonProcess.on('error', (err) => {
            console.error('Failed to start Python process:', err);
            if (!responseSent) {
                res.status(500).json({
                    success: false,
                    message: 'Failed to start face capture process',
                    error: err.message
                });
                responseSent = true;
            }
        });

    } catch (error) {
        console.error('Error in face capture:', error);
        if (!responseSent) {
            res.status(500).json({
                success: false,
                message: 'Failed to start face capture process',
                error: error.message
            });
            responseSent = true;
        }
    }
});

import test from 'node:test'
import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'

const abi = JSON.parse(readFileSync(new URL('../abi/ZKLocalVoting.json', import.meta.url), 'utf8'))
const functions = abi.filter((entry) => entry.type === 'function')
const byName = new Map(functions.map((entry) => [entry.name, entry]))

function signature(name) {
  const entry = byName.get(name)
  return entry ? `${name}(${entry.inputs.map((input) => input.type).join(',')})` : null
}

test('ZK ABI exposes candidateId/commitment/proof lifecycle', () => {
  assert.equal(signature('registerCandidate'), 'registerCandidate(uint256)')
  assert.equal(signature('registerVoterCommitment'), 'registerVoterCommitment(uint256)')
  assert.equal(signature('castVote'), 'castVote(uint256,tuple)')
  assert.equal(signature('startElection'), 'startElection()')
  assert.equal(signature('endElection'), 'endElection()')
  assert.equal(signature('getVoteCount'), 'getVoteCount(uint256)')
  assert.equal(signature('getElectionState'), 'getElectionState()')
  assert.equal(signature('getGroupRoot'), 'getGroupRoot()')
  assert.equal(signature('getGroupSize'), 'getGroupSize()')
})

test('ZK ABI has no active voterHash or civil identity transaction input', () => {

    assert.equal(serialized.includes(forbidden), false, `forbidden ABI term: ${forbidden}`)
  }

  assert.equal(castVote.inputs[1].type, 'tuple')
